function score(ans) {
    var scene0=ans.scene||[];

    if(scene0.indexOf('studio')!==-1){
        var wantDomestic=ans.region==='domestic';
        var se0=[];
        for(var k0 in DEST){
            if(Object.prototype.hasOwnProperty.call(DEST,k0)&&DEST[k0].studio&&!!DEST[k0].domestic===wantDomestic) se0.push(k0);
        }
        window._lastMaxScore=se0.length?1:0;
        return se0;
    }

    if(ans.region==='domestic'){
        return scoreDomestic(ans);
    }

    var eligible=getEligible(ans);
    var s={};
    for(var i=0;i<eligible.length;i++) s[eligible[i]]=0;

    var who=ans.who||'', scene=ans.scene||[], moment=ans.moment||[], style=ans.style||'';

    function add(ks,p){ for(var i=0;i<ks.length;i++){ if(ks[i] in s) s[ks[i]]+=p; } }
    function sub(ks,p){ for(var i=0;i<ks.length;i++){ if(ks[i] in s) s[ks[i]]-=p; } }
    var sc=0;
    for(var i=0;i<scene.length;i++){ if(scene[i]!=='studio') sc++; }
    var pd=sc>1?sc:1;
    function penExcept(pos,p){
        for(var i=0;i<eligible.length;i++){
            var k=eligible[i];
            if(pos.indexOf(k)===-1 && k in s) s[k]-=Math.ceil(p/pd);
        }
    }

    var hm=[];
    for(var i=0;i<moment.length;i++){ if(moment[i]==='starry'||moment[i]==='balloon'||moment[i]==='animal') hm.push(moment[i]); }
    var hasNone=moment.indexOf('none')!==-1;
    if(hm.length>0&&!hasNone){
        var mu=[];
        for(var i=0;i<eligible.length;i++){
            var k=eligible[i];
            if((hm.indexOf('starry')!==-1&&DEST[k]&&DEST[k].starry)||
               (hm.indexOf('balloon')!==-1&&DEST[k]&&DEST[k].balloon)||
               (hm.indexOf('animal')!==-1&&DEST[k]&&DEST[k].animal)){
                mu.push(k);
            }
        }
        if(!mu.length) return [];
        eligible=mu;
        var newS={};
        for(var i=0;i<eligible.length;i++) newS[eligible[i]]=0;
        s=newS;
    }

    var nns=[];
    for(var i=0;i<scene.length;i++){ if(scene[i]!=='none') nns.push(scene[i]); }
    if(nns.length>0){
        var anySceneOk=false;
        for(var i=0;i<nns.length;i++){
            for(var j=0;j<eligible.length;j++){
                if(DEST[eligible[j]]&&DEST[eligible[j]][nns[i]]){ anySceneOk=true; break; }
            }
            if(anySceneOk) break;
        }
        if(!anySceneOk) return [];
    }

    if(who==='couple')       add(['paris','maldives','greece','cappadocia','wien','bled','portugal','mont_saint','swiss'],3);
    if(who==='family_kids')  { add(['okinawa','guam','hawaii','bali','danang','cebu','australia'],4); sub(['paris','cappadocia','swiss','uyuni','masaimara'],3); }
    if(who==='family_elder') add(['hawaii','okinawa','guam','bali','paris','wien','italy'],3);
    if(who==='group')        add(['hawaii','okinawa','bali','guam'],3);

    var ss={beach:4,nature:4,landscape:5,street:4,castle:5,architecture:4,ruins:5,chapel:5,garden:5,villa:4};
    for(var i=0;i<scene.length;i++){
        var stype=scene[i];
        if(stype==='none'||stype==='studio') continue;
        var p=ss[stype]||4;
        var mk=[];
        for(var j=0;j<eligible.length;j++){ if(DEST[eligible[j]]&&DEST[eligible[j]][stype]) mk.push(eligible[j]); }
        add(mk,p);
        penExcept(mk,Math.ceil(p/2));
    }

    if(!hasNone){
        var ms={sunset:4,starry:5,nightview:4,balloon:8,animal:5,morning:3};
        for(var i=0;i<moment.length;i++){
            var mtype=moment[i];
            if(!ms[mtype]) continue;
            var p=ms[mtype];
            var mk=[];
            for(var j=0;j<eligible.length;j++){ if(DEST[eligible[j]]&&DEST[eligible[j]][mtype]) mk.push(eligible[j]); }
            add(mk,p);
            if(mtype==='starry'||mtype==='balloon'){
                for(var j=0;j<eligible.length;j++){
                    var k=eligible[j];
                    if(mk.indexOf(k)===-1&&k in s) s[k]-=3;
                }
            }
        }
    }

    if(style==='resort')      { add(['maldives','bali','hawaii','okinawa_rito','fiji','cancun','phuket','cebu'],4); sub(['paris','wien','central_eu','romantic_road','netherlands','egypt','morocco','newyork','london'],1); }
    if(style==='sightseeing') { add(['paris','wien','italy','barcelona','cappadocia','nz','australia','egypt','central_eu','london','portugal','morocco','romantic_road'],4); sub(['okinawa','okinawa_rito','maldives','fiji','bali','guam','cebu'],1); }
    if(style==='shopping')    add(['paris','italy','bali','hawaii','korea','singapore','newyork','london','barcelona'],4);
    if(style==='active')      { add(['nz','australia','canada','masaimara','cancun','hokkaido','cappadocia'],4); sub(['maldives','paris','wien','central_eu'],1); }

    // Penalty: Europe destinations when days=long (short schedule)
    var euroKeys=['paris','wien','italy','barcelona','greece','central_eu','romantic_road','netherlands','london','dubrovnik','bled','portugal','malta','cappadocia','nordic','mont_saint'];
    if(ans.days==='long'){
        var hasEuro=false;
        for(var i=0;i<eligible.length;i++){ if(euroKeys.indexOf(eligible[i])!==-1){hasEuro=true;break;} }
        if(hasEuro) sub(euroKeys,2);
    }

    var pairs=[];
    for(var k in s){ if(Object.prototype.hasOwnProperty.call(s,k)) pairs.push([k,s[k]]); }
    pairs.sort(function(a,b){return b[1]-a[1];});
    window._lastMaxScore=pairs.length>0?pairs[0][1]:0;
    var result=[];
    for(var i=0;i<pairs.length;i++) result.push(pairs[i][0]);
    return result;
}

function calcMatching(key,ans) {
    var d=DEST[key]; if(!d) return {rate:0,matched:[],unmatched:[]};
    var scene=ans.scene||[], moment=ans.moment||[], style=ans.style||'';
    var SL={beach:'ビーチ・海',nature:'大自然・森',landscape:'奇岩・砂漠',street:'街並み・路地裏',castle:'お城・宮殿',architecture:'美術館・歴史建築',ruins:'遺跡',chapel:'チャペル・教会',garden:'花界・庭園',villa:'ヴィラ・ホテル',studio:'スタジオ'};
    var ML={sunset:'サンセット',starry:'星空',nightview:'夜景・イルミネーション',balloon:'気球・バルーン',animal:'動物との撮影',morning:'朝日・早朝の光'};
    var resortList=['maldives','bali','hawaii','okinawa_rito','fiji','cancun','phuket','cebu'];
    var sightList=['paris','wien','italy','barcelona','cappadocia','nz','australia','egypt','central_eu','london','portugal','morocco','romantic_road'];
    var shopList=['paris','italy','bali','hawaii','korea','singapore','newyork','london','barcelona'];
    var activeList=['nz','australia','canada','masaimara','cancun','hokkaido','cappadocia'];
    var styleMatch={
        resort:function(){ return resortList.indexOf(key)!==-1; },
        sightseeing:function(){ return sightList.indexOf(key)!==-1; },
        shopping:function(){ return shopList.indexOf(key)!==-1; },
        active:function(){ return activeList.indexOf(key)!==-1; }
    };
    var SML={resort:'のんびりリゾート',sightseeing:'観光めぐり',shopping:'グルメ・ショッピング',active:'アクティブ・冒険'};
    var conds=[];
    for(var i=0;i<scene.length;i++){
        var x=scene[i]; if(x==='none') continue;
        if(SL[x]) conds.push({label:SL[x],matched:!!d[x],type:'scene'});
    }
    for(var i=0;i<moment.length;i++){
        var x=moment[i]; if(x==='none') continue;
        if(ML[x]) conds.push({label:ML[x],matched:!!d[x],type:'moment'});
    }
    if(style&&styleMatch[style]) conds.push({label:SML[style]||style,matched:styleMatch[style](),type:'style'});
    var matched=[],unmatched=[];
    for(var i=0;i<conds.length;i++){
        if(conds[i].matched) matched.push(conds[i]);
        else unmatched.push(conds[i]);
    }
    var rate=conds.length>0?Math.round(matched.length/conds.length*100):100;
    return {rate:rate,matched:matched,unmatched:unmatched};
}

                setTimeout(function(){if(curQIdx<getTotalSteps()-1){curQIdx++;renderQ('right');}else showLoading();},300);
            });
            $s.appendChild(btn);
        });
    } else {
        $s.style.display='none'; $m.style.display='flex'; $n.style.display='block'; $n.disabled=true; $m.innerHTML='';
        var qi=qFlow[curQIdx];
        if(!multiSel[qi]){multiSel[qi]={}; var _ia=answers[q.id]||[]; for(var _ii=0;_ii<_ia.length;_ii++) multiSel[qi][_ia[_ii]]=true;}
        var hiddenVals={};
        if(answers.region==='domestic'){
            if(q.id==='scene')  hiddenVals={castle:true,architecture:true,villa:true};
            if(q.id==='moment') hiddenVals={balloon:true,animal:true};
        }
        var visibleOpts=q.opts.filter(function(o){return !hiddenVals[o.v];});
        var specialMoment={starry:true,balloon:true,animal:true};
        var btnMap={};
        function refreshMultiState(){
            var selectedSpecial=false;
            if(q.id==='moment'){
                for(var sv in specialMoment){ if(Object.prototype.hasOwnProperty.call(specialMoment,sv)&&multiSel[qi][sv]){selectedSpecial=true;break;} }
            }
            visibleOpts.forEach(function(opt){
                var btn=btnMap[opt.v];
                var dis=q.id==='moment'&&selectedSpecial&&!specialMoment[opt.v]&&opt.v!=='none';
                btn.disabled=dis;
                btn.classList.toggle('opt-disabled',dis);
                btn.classList.toggle('selected',!!multiSel[qi][opt.v]);
            });
            $n.disabled=Object.keys(multiSel[qi]||{}).length===0;
        }
        visibleOpts.forEach(function(opt){
            var btn=document.createElement('button');
            btn.className='opt-multi'+(multiSel[qi][opt.v]?' selected':'');
            btn.innerHTML='<span class="chk"></span><span class="opt-multi-text">'+opt.text+'</span>';
            btn.addEventListener('click',function(){
                if(btn.disabled) return;
                if(multiSel[qi][opt.v]){delete multiSel[qi][opt.v];}
                else{
                    if(opt.v==='none'){multiSel[qi]={};}
                    else{
                        delete multiSel[qi]['none'];
                        if(q.id==='moment'&&specialMoment[opt.v]){
                            for(var k in multiSel[qi]){ if(Object.prototype.hasOwnProperty.call(multiSel[qi],k)&&!specialMoment[k]) delete multiSel[qi][k]; }
                        }
                    }
                    multiSel[qi][opt.v]=true;
                }
                refreshMultiState();
                if(q.id==='scene') updateFlow();
            });
            $m.appendChild(btn);
            btnMap[opt.v]=btn;
        });
        refreshMultiState();
    }
    showScreen('screen-q',dir==='right'?'anim-right':'anim-left');
}

function showLoading(){
    showScreen('screen-loading','anim-up');
    var ranked=score(answers), origMax=window._lastMaxScore;
    var allNames=Object.values(DEST).map(function(d){return d.name;});
    var si=document.getElementById('slot-inner'), fe=document.getElementById('loading-final');
    fe.style.display='none'; fe.classList.remove('visible');
    var items=allNames.concat(allNames.slice(0,10),(DEST[ranked[0]]?DEST[ranked[0]].name:''));
    si.innerHTML=items.map(function(n){return '<div class="loading-slot-item">'+n+'</div>';}).join('');
    var idx=0,speed=80,count=0,max=items.length-1;
    function spin(){
        si.style.transform='translateY(-'+(idx*60)+'px)';
        idx++; count++;
        speed=count<max*.6?80:count<max*.85?120:200;
        if(count<max){setTimeout(spin,speed);}
        else{setTimeout(function(){
            document.querySelector('.loading-slot').style.display='none';
            fe.style.display='block'; fe.textContent=(DEST[ranked[0]]?DEST[ranked[0]].name:'');
            setTimeout(function(){fe.classList.add('visible');},50);
            setTimeout(function(){(function(){
    var dbg='start';
    try{
        dbg='score'; 
        dbg='calcMatch';
        dbg='renderResult';
        renderResult(ranked,origMax);
    }catch(e){
        alert('Step:'+dbg+' Err:'+e.message+' '+(e.stack?e.stack.split('n')[1]:''));
    }
})()},1200);
        },300);}
    }
    setTimeout(spin,400);
}

    el.style.display='block'; el.classList.add('active');

}
