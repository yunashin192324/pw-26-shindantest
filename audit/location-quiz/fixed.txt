            for(var j=0;j<eligible.length;j++){ if(hasFeature(eligible[j],mtype)) mk.push(eligible[j]); }
            add(mk,p);

                for(var j=0;j<eligible.length;j++){
                    var k=eligible[j];
                    if(mk.indexOf(k)===-1&&k in s) s[k]-=3;
                }
            }
        }
    }

    if(style==='resort')      { add(['okinawa','okinawa_rito','guam','maldives','bali','hawaii','fiji','cancun','phuket','cebu','danang','malta','greece'],4); sub(['paris','wien','central_eu','romantic_road','netherlands','egypt','morocco','newyork','london'],1); }
    if(style==='sightseeing') { add(['hokkaido','korea','danang','singapore','australia','lasvegas','newyork','canada','dubai','egypt','morocco','masaimara','uyuni','paris','mont_saint','wien','italy','barcelona','portugal','malta','greece','cappadocia','swiss','central_eu','romantic_road','netherlands','dubrovnik','bled','nordic','london'],4); sub(['okinawa','okinawa_rito','maldives','fiji','bali','guam','cebu'],1); }
    if(style==='shopping')    add(['okinawa','korea','guam','bali','singapore','hawaii','newyork','dubai','paris','italy','barcelona','london'],4);
    if(style==='active')      { add(['okinawa_rito','hokkaido','australia','lasvegas','canada','cancun','nz','masaimara','uyuni','cappadocia','swiss','nordic'],4); sub(['maldives','paris','wien','central_eu'],1); }

        if(hasEuro) sub(euroKeys,2);

    for(var k in s){ if(Object.prototype.hasOwnProperty.call(s,k)) pairs.push([k,s[k]]); }

        var k = pairs[i][0];
        if(DEST[k] && !!DEST[k].domestic === wantDomestic){

        }
    }
    // 並べ替えの優先順位：①選んだ星空・気球・動物を多く満たす ②選んだ景色を1つ以上満たす ③一致率（画面に表示する順位と一致率を矛盾させない） ④点数 ⑤同点なら選んだ予算区分に登録された撮影地
    var nsc=[]; for(var i=0;i<scene.length;i++){ if(scene[i]!=='none') nsc.push(scene[i]); }
    function rankKey(k){
        var h=0; if(!hasNone){ for(var i=0;i<hm.length;i++){ if(hasFeature(k,hm[i])) h++; } }
        var sh=0; for(var i=0;i<nsc.length;i++){ if(hasFeature(k,nsc[i])){ sh=1; break; } }

        return [h,sh,ib,calcMatching(k,ans).rate];
    }
    for(var i=0;i<finalPairs.length;i++) finalPairs[i].push(rankKey(finalPairs[i][0]));
    finalPairs.sort(function(a,b){ return (b[2][0]-a[2][0])||(b[2][1]-a[2][1])||(b[2][3]-a[2][3])||(b[1]-a[1])||(b[2][2]-a[2][2]); });
    window._lastMaxScore=finalPairs.length>0?finalPairs[0][1]:0;
   
    var result=[];
    for(var i=0;i<finalPairs.length;i++) result.push(finalPairs[i][0]);
    return result;
}



function calcMatching(key,ans) {
    var d=DEST[key]; if(!d) return {rate:0,matched:[],unmatched:[]};
    var scene=ans.scene||[], moment=ans.moment||[], style=ans.style||'';

    var SL={beach:'ビーチ・海',nature:'大自然・森',landscape:'奇岩・砂漠',street:'街並み・路地裏',castle:'お城・宮殿',architecture:'美術館・歴史建築',ruins:'遺跡',chapel:'チャペル・教会',garden:'花畑・庭園',villa:'ヴィラ・ホテル',studio:'スタジオ'};
    var ML={sunset:'サンセット',starry:'星空',nightview:'夜景・イルミネーション',balloon:'気球・バルーン',animal:'動物との撮影',morning:'朝日・早朝の光'};
    var SML={resort:'のんびりリゾート',sightseeing:'観光めぐり',shopping:'グルメ・ショッピング',active:'アクティブ・冒険'};

    // 1. 景色・ロケーション（scene）、2. 特別な瞬間（moment）の対応判定は hasFeature()（SCENE_LIST / MOMENT_LIST）に統一

    // 3. 旅行スタイル（style）に対応しているエリア一覧
    var resortList=['okinawa','okinawa_rito','guam','maldives','bali','hawaii','fiji','cancun','phuket','cebu','danang','malta','greece'];
    var sightList=['hokkaido','korea','danang','singapore','australia','lasvegas','newyork','canada','dubai','egypt','morocco','masaimara','uyuni','paris','mont_saint','wien','italy','barcelona','portugal','malta','greece','cappadocia','swiss','central_eu','romantic_road','netherlands','dubrovnik','bled','nordic','london'];
    var shopList=['okinawa','korea','guam','bali','singapore','hawaii','newyork','dubai','paris','italy','barcelona','london'];
    var activeList=['okinawa_rito','hokkaido','australia','lasvegas','canada','cancun','nz','masaimara','uyuni','cappadocia','swiss','nordic'];

    var styleMatch={
        resort:function(){ return resortList.indexOf(key)!==-1; },
        sightseeing:function(){ return sightList.indexOf(key)!==-1; },
        shopping:function(){ return shopList.indexOf(key)!==-1; },
        active:function(){ return activeList.indexOf(key)!==-1; }
    };

    var conds=[];


    for(var i=0;i<scene.length;i++){
        var x=scene[i]; if(x==='none') continue;
        if(SL[x]){
            var isSceneMatch = hasFeature(key,x);
            conds.push({label:SL[x], matched:isSceneMatch, type:'scene'});
        }
    }

 
    for(var i=0;i<moment.length;i++){
        var x=moment[i]; if(x==='none') continue;
        if(ML[x]){
            var isMomentMatch = hasFeature(key,x);
            conds.push({label:ML[x], matched:isMomentMatch, type:'moment'});
        }
    }


    if(style&&styleMatch[style]) conds.push({label:SML[style]||style,matched:styleMatch[style](),type:'style'});

    var matched=[],unmatched=[];
    for(var i=0;i<conds.length;i++){
        if(conds[i].matched) matched.push(conds[i]);
        else unmatched.push(conds[i]);
    }
    var rate=conds.length>0?Math.round(matched.length/conds.length*100):100;
    return {rate:rate,matched:matched,unmatched:unmatched};
}

function showScreen(id,anim){
    anim=anim||'anim-right';
    document.querySelectorAll('.screen').forEach(function(s){s.classList.remove('active');s.style.display='none';});
    var el=document.getElementById(id);
    el.style.display='block'; el.classList.add('active');
    el.classList.remove('anim-right','anim-left','anim-up');
    void el.offsetWidth; el.classList.add(anim);
    if(id==='screen-loading'){document.querySelector('.loading-slot').style.display='block';document.getElementById('loading-final').style.display='none';document.getElementById('loading-final').classList.remove('visible');}
window.scrollTo({top:0,behavior:'instant'});


    document.querySelectorAll('.bg-image').forEach(function(bg){ bg.classList.remove('active'); });
   

    if(id === 'screen-start') {
        document.getElementById('bg-start').classList.add('active');
    } else if (id === 'screen-q') {
        var stepId = 'bg-q' + (curQIdx + 1);
        var bgEl = document.getElementById(stepId);
        if(bgEl) bgEl.classList.add('active');
        else document.getElementById('bg-q1').classList.add('active');
    } else if (id === 'screen-loading' || id === 'screen-result') {
        var resultBg = document.getElementById('bg-result');
        if(resultBg) resultBg.classList.add('active');
    } else {
        var lastBg = document.getElementById('bg-q' + (curQIdx + 1));
        if(lastBg) lastBg.classList.add('active');
        else document.getElementById('bg-start').classList.add('active');
    }
}
